# 01: Pymunkとは!?

[Pymunk](https://www.pymunk.org/en/latest/)は、Chipmunkをベースに作られた、2Dゲーム用物理ライブラリです。
今回は、本ライブラリの導入から簡単な動作確認まで、できるだけ簡潔にまとめていきます。

![](/images/articles/815885c2c48d12/01.png)
*Pymunk*

Pymunkは、物理上の物体の形や角度、移動速度、衝突処理の計算だけを行うライブラリです(描画は一切行いません)。
よって、空間に存在する物体の描画部分は実装側で用意しなくてはいけません。

ここでは、描画部分を[Pyxel](https://github.com/kitao/pyxel)で実装します。
Pyxelは、激かわレトロゲームが作れる神フレームワークです。
使い方については、[PythonでPyxelをかじる本](https://zenn.dev/sdkfz181tiger/books/675e49a3e4d34b)をご参照ください。

![](/images/articles/815885c2c48d12/02.png)
*Pyxel*

# 02: インストール

```command:Pymunkインストール

```

```command:Pyxelインストール

```

# 03: Pymunkの基本

Pymunkは、2D空間にある全ての物体の物理計算を一手に引き受けてくれるライブラリです。

先ず最初に、Pymunkが管理する物理空間を```pymunk.Space()```で用意し、重力の方向を指定します。

```python:main.py(抜粋)
self.space = pymunk.Space() # 物理空間を用意する
self.space.gravity = 0, 820 # 重力の方向(x, y)を指定する(Pyxelではyは+)
```

次に、物理空間を更新する処理を```update()```内に記述します。
複雑な物理的計算はこの関数内部で一括で行ってくれます。

```python:main.py(抜粋)
self.space.step(DELAY_TIME) # Step
```

# 04: 計算結果と描画処理を橋渡しするクラス

Pymunkには物理計算と描画処理との橋渡しとして、```pymunk.SpaceDebugDrawOptions```クラスが用意されています。

そこで今回は、本クラスを継承したPyxel向けのクラスを実装しました。(今回のポイントです)
Pyxelで使える様に継承した例は次のとおりです。(ここでは、全て"白"のラインで統一)

```python:main.py
COLOR = 7 # White

class SimpleDebugDrawOptions(pymunk.SpaceDebugDrawOptions):
    """ DebugDrawOptions for Pyxel """

    def __init__(self, camera=(0, 0)):
        super().__init__()
        # Camera
        self.camera = pymunk.Vec2d(*camera)

    def p(self, v):
        """カメラ座標に変換する処理"""
        return (int(v.x - self.camera.x), int(v.y - self.camera.y))

    def draw_circle(self, pos, angle, radius, outline_color, fill_color):
        """円を描く処理"""
        pyxel.circb(int(pos.x), int(pos.y), radius, COLOR)
        end_x = pos.x + math.cos(angle) * radius
        end_y = pos.y + math.sin(angle) * radius
        pyxel.line(pos.x, pos.y, end_x, end_y, COLOR)

    def draw_segment(self, a, b, color):
        """線を引く処理"""
        pyxel.line(int(a.x), int(a.y), int(b.x), int(b.y), COLOR)

    def draw_fat_segment(self, a, b, radius, outline_color, fill_color):
        """線を引く処理"""
        pyxel.line(int(a.x), int(a.y), int(b.x), int(b.y), COLOR)

    def draw_polygon(self, verts, radius, outline_color, fill_color):
        """多角形を描く処理"""
        n = len(verts)
        if n < 2: return
        for i in range(n):
            v1 = verts[i]
            v2 = verts[(i + 1) % n]
            pyxel.line(v1.x, v1.y, v2.x, v2.y, COLOR)

    def draw_dot(self, size, pos, color):
    	"""点を描く処理"""
        pyxel.pset(pos.x, pos.y, COLOR)
```

使い方はとてもシンプルです。
最初にコンストラクタで、上記クラスをインスタンス化します。

```python:main.py(抜粋)
self.ddo = SimpleDebugDrawOptions() # DebugDraw
```

あとは、描画部分で```debug_draw()```に渡すだけです。

```python:main.py(抜粋)
self.space.debug_draw(self.ddo) # DebugDraw
```

# 05: Pyxelと組み合わせる

PyxelとPymunkを組み合わせると次の様になります。

:::details 完成コード(main.py)
```python:main.py
import pymunk
import pyxel

W, H       = 320, 320
FPS        = 32
DELAY_TIME = 1 / FPS
COLOR      = 7 # White

# PyxelDrawOptions
class SimpleDebugDrawOptions(pymunk.SpaceDebugDrawOptions):
    """ DebugDrawOptions for Pyxel """

    def __init__(self, camera=(0, 0)):
        super().__init__()
        # Camera
        self.camera = pymunk.Vec2d(*camera)

    def p(self, v):
        """カメラ座標に変換する処理"""
        return (int(v.x - self.camera.x), int(v.y - self.camera.y))

    def draw_circle(self, pos, angle, radius, outline_color, fill_color):
        """円を描く処理"""
        pyxel.circb(int(pos.x), int(pos.y), radius, COLOR)
        end_x = pos.x + math.cos(angle) * radius
        end_y = pos.y + math.sin(angle) * radius
        pyxel.line(pos.x, pos.y, end_x, end_y, COLOR)

    def draw_segment(self, a, b, color):
        """線を引く処理"""
        pyxel.line(int(a.x), int(a.y), int(b.x), int(b.y), COLOR)

    def draw_fat_segment(self, a, b, radius, outline_color, fill_color):
        """線を引く処理"""
        pyxel.line(int(a.x), int(a.y), int(b.x), int(b.y), COLOR)

    def draw_polygon(self, verts, radius, outline_color, fill_color):
        """多角形を描く処理"""
        n = len(verts)
        if n < 2: return
        for i in range(n):
            v1 = verts[i]
            v2 = verts[(i + 1) % n]
            pyxel.line(v1.x, v1.y, v2.x, v2.y, COLOR)

    def draw_dot(self, size, pos, color):
    	"""点を描く処理"""
        pyxel.pset(pos.x, pos.y, COLOR)

# Game
class Game:
    def __init__(self):
        """ Constructor """

        # Pymunk
        self.space = pymunk.Space()
        self.space.gravity = 0, 820
        self.ddo = SimpleDebugDrawOptions() # DebugDraw

        # Segment(地面を追加する)
        points = [
                (45, H - 120),
                (50, H - 70),
                (W - 50, H - 75)]
        for a, b in zip(points, points[1:]):
            ground = pymunk.Segment(
                self.space.static_body, a, b, 4)
            ground.friction = 1.0
            self.space.add(ground) # Add

        # Box, Circle(箱や円を追加する)
        self.create_box(W/2-30,    H/2-64, 16, 32)
        self.create_box(W/2,       H/2-48, 32, 16)
        self.create_circle(W/2,    H/2-64, 8)
        self.create_circle(W/2+48, H/2-96, 16)

        # Pyxel
        pyxel.init(W, H, title="Hello, Pyxel!!", fps=FPS)
        pyxel.run(self.update, self.draw)

    def update(self):
        self.space.step(DELAY_TIME) # Step

    def draw(self):
        # Clear
        pyxel.cls(0)
        self.space.debug_draw(self.ddo) # DebugDraw

    def create_box(self, x, y, w, h):
        body = pymunk.Body()
        body.position = (x, y)
        shape = pymunk.Poly.create_box(body, (w, h))
        shape.mass = 10
        self.space.add(body, shape)

    def create_circle(self, x, y, r):
        body = pymunk.Body()
        body.position = (x, y)
        shape = pymunk.Circle(body, r)
        shape.mass = 10
        self.space.add(body, shape)

def main():
    """ Main """
    Game()

if __name__ == "__main__":
    main()
```

# 07: 物理空間に物体を追加する

完成コードに記述済みですが、物理空間に物体を追加するパターンを簡単に紹介します。

基本的には、```pymunk.Body()```で作られるBodyオブジェクトには座標を、
```pymunk.Poly.create_box()```で作られるShapeオブジェクトは形を指定します。
最後に、```space.add()```で物理空間に物体を追加します。

```python:main.py(四角)
body = pymunk.Body()
body.position = (x, y)
shape = pymunk.Poly.create_box(body, (w, h))
shape.mass = 10
self.space.add(body, shape)
```

円も同じ様に作ることができます。

```python:main.py(円)
body = pymunk.Body()
body.position = (x, y)
shape = pymunk.Circle(body, r)
shape.mass = 10
self.space.add(body, shape)
```

# 08: ジョイントで人形を作る

物体と物体を接続(ジョイント)して、人形(RagDoll)を作ってみます。
ここでは単純に、```pymunk.PinJoint()```を使って接続します。

:::details 完成コード(main.py)
```python:main.py(抜粋)
import pymunk
import pyxel
import math
import random

W, H       = 320, 240
FPS        = 32
DELAY_TIME = 1 / FPS
COLOR      = 7 # White

# PyxelDrawOptions
class SimpleDebugDrawOptions(pymunk.SpaceDebugDrawOptions):
    """ DebugDrawOptions for Pyxel """

    def __init__(self, camera=(0, 0)):
        super().__init__()
        # Camera
        self.camera = pymunk.Vec2d(*camera)

    def p(self, v):
        return (int(v.x - self.camera.x), int(v.y - self.camera.y))

    def draw_circle(self, pos, angle, radius, outline_color, fill_color):
        pyxel.circb(int(pos.x), int(pos.y), radius, COLOR)
        end_x = pos.x + math.cos(angle) * radius
        end_y = pos.y + math.sin(angle) * radius
        pyxel.line(pos.x, pos.y, end_x, end_y, COLOR)

    def draw_segment(self, a, b, color):
        pyxel.line(int(a.x), int(a.y), int(b.x), int(b.y), COLOR)

    def draw_fat_segment(self, a, b, radius, outline_color, fill_color):
        pyxel.line(int(a.x), int(a.y), int(b.x), int(b.y), COLOR)

    def draw_polygon(self, verts, radius, outline_color, fill_color):
        n = len(verts)
        if n < 2: return
        for i in range(n):
            v1 = verts[i]
            v2 = verts[(i + 1) % n]
            pyxel.line(v1.x, v1.y, v2.x, v2.y, COLOR)

    def draw_dot(self, size, pos, color):
        pyxel.pset(pos.x, pos.y, COLOR)

# Game
class Game:
    def __init__(self):
        """ Constructor """

        # Pymunk
        self.space = pymunk.Space()
        self.space.gravity = 0, 410 # 820
        self.ddo = SimpleDebugDrawOptions() # DebugDraw

        # Segment
        points = [
                (45, H - 120),
                (50, H - 70),
                (W - 50, H - 75)]
        for a, b in zip(points, points[1:]):
            ground = pymunk.Segment(
                self.space.static_body, a, b, 4)
            ground.friction = 1.0
            self.space.add(ground) # Add

        # RagDoll
        self.create_doll(W/2, H/2-96)

        # Pyxel
        pyxel.init(W, H, title="Hello, Pyxel!!", fps=FPS)
        pyxel.run(self.update, self.draw)

    def update(self):
        self.space.step(DELAY_TIME) # Step

    def draw(self):
        # Clear
        pyxel.cls(0)
        self.space.debug_draw(self.ddo) # DebugDraw

    def create_box(self, x, y, w, h):
        body = pymunk.Body()
        body.position = (x, y)
        shape = pymunk.Poly.create_box(body, (w, h))
        shape.mass = 10
        self.space.add(body, shape)
        return body

    def create_circle(self, x, y, r):
        body = pymunk.Body()
        body.position = (x, y)
        shape = pymunk.Circle(body, r)
        shape.mass = 10
        self.space.add(body, shape)
        return body

    def create_doll(self, x, y):
        head   = self.create_circle(x, y, 12)
        body_a = self.create_box(x, y+28, 16, 32)
        self.pin(head, body_a, (0, 8), (0, -16))

        arm_l_a = self.create_box(x-16, y+20, 8, 16)
        arm_l_b = self.create_box(x-16, y+36, 8, 16)
        self.pin(body_a, arm_l_a, (-8, -16), (0, -8))
        self.pin(arm_l_a, arm_l_b, (0, 8), (0, -8))

        arm_r_a = self.create_box(x+16, y+20, 8, 16)
        arm_r_b = self.create_box(x+16, y+36, 8, 16)
        self.pin(body_a, arm_r_a, (8, -16), (0, -8))
        self.pin(arm_r_a, arm_r_b, (0, 8), (0, -8))

        leg_l_a = self.create_box(x-8, y+52, 8, 16)
        leg_l_b = self.create_box(x-8, y+68, 8, 16)
        self.pin(body_a, leg_l_a, (-8, 16), (0, -8))
        self.pin(leg_l_a, leg_l_b, (0, 8), (0, -8))

        leg_r_a = self.create_box(x+8, y+52, 8, 16)
        leg_r_b = self.create_box(x+8, y+68, 8, 16)
        self.pin(body_a, leg_r_a, (+8, 16), (0, -8))
        self.pin(leg_r_a, leg_r_b, (0, 8), (0, -8))

    def pin(self, body_a, body_b, anc_a, anc_b):
        joint = pymunk.PinJoint(body_a, body_b, anc_a, anc_b)
        self.space.add(joint)

def main():
    """ Main """
    Game()

if __name__ == "__main__":
    main()
```
